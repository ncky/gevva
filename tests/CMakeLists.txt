cmake_minimum_required(VERSION 3.28)
# These regression tests can also be built without CUDA: cmake -S tests -B build/cpu-tests
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(gevva-cpu-tests LANGUAGES CXX)
  include(CTest)
endif()
find_package(nlohmann_json 3.11 REQUIRED)
add_executable(gevva-safetensors-test test_safetensors.cpp ../src/safetensors.cpp)
target_compile_features(gevva-safetensors-test PRIVATE cxx_std_20)
target_include_directories(gevva-safetensors-test PRIVATE ../include)
target_link_libraries(gevva-safetensors-test PRIVATE nlohmann_json::nlohmann_json)
add_test(NAME safetensors_validation COMMAND gevva-safetensors-test)
add_executable(gevva-fp8-resources-test test_fp8_resources.cpp)
target_compile_features(gevva-fp8-resources-test PRIVATE cxx_std_20)
target_include_directories(gevva-fp8-resources-test PRIVATE ../src)
add_test(NAME fp8_resource_failure_cleanup COMMAND gevva-fp8-resources-test)
